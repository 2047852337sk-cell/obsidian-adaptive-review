# 复习完成归档

> 新卡片通过最后阶段后会自动进入归档；旧卡片完成全部复选框后也会继续显示。点击「重新开始」会保留既有复习历史，并从今天开始新一轮。

---

```dataviewjs
const { Notice } = require('obsidian');
const intervalKeys = ['D0', 'D1', 'D3', 'D7', 'D14', 'D30', 'D60'];
const legacyKeys = ['D0', 'D1', 'D7', 'D14', 'D30', 'D60'];
const pages = dv.pages('#复习/待复习').where(p =>
  (p.复习起始日 || p.学习日期) &&
  !p.file.folder.includes('模板') &&
  p.file.path !== dv.current().file.path &&
  p.file.name !== '复习总表'
);

const pad = n => String(n).padStart(2, '0');
const fmtFull = d => `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}`;
const asArray = value => {
  if (!value) return [];
  if (Array.isArray(value)) return value;
  if (typeof value.array === 'function') return value.array();
  return [value];
};
const escapeRegExp = text => text.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');

const getChecked = content => {
  const checked = {};
  if (content) {
    const matches = content.matchAll(/-\s*\[([ xX])\]\s*(D\d+)/g);
    for (const match of matches) checked[match[2]] = match[1].toLowerCase() === 'x';
  }
  return checked;
};

const getCategory = p => {
  const explicit = asArray(p.分类).map(value => String(value).trim()).filter(Boolean);
  if (explicit.length > 0) return explicit[0];
  const tags = asArray(p.file.etags ?? p.file.tags)
    .map(value => String(value).replace(/^#/, '').trim())
    .filter(tag => tag && tag !== '复习/待复习' && tag !== '复习' && !tag.startsWith('复习/'));
  return tags[0] ?? '未分类';
};

const frontmatterValue = (content, key, fallback = '') => {
  const match = content.match(/^---\r?\n([\s\S]*?)\r?\n---/);
  if (!match) return fallback;
  const field = match[1].match(new RegExp(`^${escapeRegExp(key)}:[ \\t]*(.*)$`, 'm'));
  return field ? field[1].trim().replace(/^['"]|['"]$/g, '') : fallback;
};

const upsertFrontmatter = (content, key, value) => {
  const match = content.match(/^---\r?\n[\s\S]*?\r?\n---/);
  if (!match) return content;
  let frontmatter = match[0].replace(/\r?\n---$/, '');
  const rest = content.slice(match[0].length);
  const pattern = new RegExp(`^${escapeRegExp(key)}:.*$`, 'm');
  frontmatter = pattern.test(frontmatter)
    ? frontmatter.replace(pattern, `${key}: ${value ?? ''}`)
    : `${frontmatter}\n${key}: ${value ?? ''}`;
  return `${frontmatter}\n---${rest}`;
};

const clearReviewTextValue = (content, key) => content.replace(
  new RegExp(`^${escapeRegExp(key)}(?:::|：).*$`, 'm'),
  `${key}：`
);

const appendHistory = (content, entry) => {
  const marker = ['<!-- REVIEW_HISTORY', '_END -->'].join('');
  const markerIndex = content.lastIndexOf(marker);
  return markerIndex >= 0
    ? `${content.slice(0, markerIndex)}${entry}\n${content.slice(markerIndex)}`
    : `${content.trimEnd()}\n\n## 复习历史\n\n${entry}\n${marker}\n`;
};

const resetLegacyRecords = content => content
  .replace(/-\s*\[[ xX]\]\s*(D0|D1|D3|D7|D14|D30|D60)/g, '- [ ] $1')
  .replace(
    /-\s*\[[ xX]\]\s*([1-4]\.\s*(?:完全不会|有印象但不完整|基本掌握|熟练掌握)[^\r\n]*)/g,
    '- [ ] $1'
  );

const restartAdaptive = async (p, button) => {
  const ok = confirm(
    `确定要把「${p.file.name}」从今天重新开始吗？\n\n原复习历史会保留；阶段回到 D0，轮次加 1，下次复习日设为今天。`
  );
  if (!ok) return;
  const file = app.vault.getAbstractFileByPath(p.file.path);
  if (!file) {
    new Notice('没有找到这篇笔记，请先打开检查一下');
    return;
  }
  const today = new Date();
  today.setHours(0, 0, 0, 0);
  const dateText = fmtFull(today);
  let content = await app.vault.read(file);
  const nextRound = (Number(frontmatterValue(content, '复习轮次', '1')) || 1) + 1;
  content = upsertFrontmatter(content, '复习起始日', dateText);
  content = upsertFrontmatter(content, '当前阶段', 'D0');
  content = upsertFrontmatter(content, '下次复习日', dateText);
  content = upsertFrontmatter(content, '连续熟练次数', 0);
  content = upsertFrontmatter(content, '复习轮次', nextRound);
  content = upsertFrontmatter(content, '复习模式', '完整复习');
  content = upsertFrontmatter(content, '最后评分', '');
  content = upsertFrontmatter(content, '最近复习日', '');
  content = upsertFrontmatter(content, '状态', '进行中');
  content = clearReviewTextValue(content, '本轮最大漏洞');
  content = clearReviewTextValue(content, '一句话纠正');
  content = appendHistory(
    content,
    `### ${dateText} · 手动重新开始\n\n- 新轮次：第 ${nextRound} 轮\n- 新阶段：阶段 1（基准 D0）\n- 下次安排：${dateText} · 完整复习`
  );
  await app.vault.modify(file, content);
  button.textContent = '已重新开始';
  button.disabled = true;
  new Notice(`已重新开始：${p.file.name}`);
};

const restartLegacy = async (p, button) => {
  const ok = confirm(
    `确定要把旧卡片「${p.file.name}」重新开始吗？\n\n复习起始日会改成今天，并清空原有节点和效果勾选。`
  );
  if (!ok) return;
  const file = app.vault.getAbstractFileByPath(p.file.path);
  if (!file) {
    new Notice('没有找到这篇笔记，请先打开检查一下');
    return;
  }
  const today = new Date();
  today.setHours(0, 0, 0, 0);
  let content = await app.vault.read(file);
  content = upsertFrontmatter(content, '复习起始日', fmtFull(today));
  content = upsertFrontmatter(content, '状态', '进行中');
  content = resetLegacyRecords(content);
  await app.vault.modify(file, content);
  button.textContent = '已重新开始';
  button.disabled = true;
  new Notice(`已重新开始：${p.file.name}`);
};

const groups = {};
for (const p of pages) {
  const adaptive = String(p.调度类型 ?? '') === '自适应';
  let completed = adaptive && String(p.状态 ?? '') === '已完成';
  if (!adaptive) {
    const content = await dv.io.load(p.file.path);
    const checked = getChecked(content);
    const hasD3 = /-\s*\[[ xX]\]\s*D3\b/.test(content ?? '');
    const required = hasD3 ? intervalKeys : legacyKeys;
    completed = required.every(key => checked[key]);
  }
  if (!completed) continue;

  const category = getCategory(p);
  const button = document.createElement('button');
  button.textContent = '重新开始';
  button.onclick = () => adaptive ? restartAdaptive(p, button) : restartLegacy(p, button);
  const row = adaptive
    ? [p.file.link, '自适应', Number(p.复习轮次 ?? 1), p.最近复习日 ?? '—', button]
    : [p.file.link, '旧卡兼容', '—', p.复习起始日 ?? p.学习日期, button];
  if (!groups[category]) groups[category] = [];
  groups[category].push(row);
}

const categories = Object.keys(groups).sort((a, b) => a.localeCompare(b));
if (categories.length === 0) {
  dv.paragraph('还没有完成整轮的知识点');
} else {
  const total = categories.reduce((sum, category) => sum + groups[category].length, 0);
  dv.paragraph(`共 ${total} 条已完成知识点，按分类整理如下。`);
  for (const category of categories) {
    const rows = groups[category].sort((a, b) => a[0].toString().localeCompare(b[0].toString()));
    dv.header(2, `${category}（${rows.length}）`);
    dv.table(['知识点', '类型', '完成轮次', '最近完成／起始日', '操作'], rows);
  }
}
```

## 分类方法

- 优先读取笔记属性里的 `分类` 字段，例如 `分类: 英语/新概念`
- 如果没有 `分类`，读取除 `复习/待复习` 外的第一个额外标签
- 如果两者都没有，归入「未分类」

## 重新开始会发生什么

- 自适应卡：保留全部历史，轮次加 1，从今天的 D0 完整复习重新开始
- 旧卡片：保留旧格式，清空节点与效果勾选，并把复习起始日改为今天
